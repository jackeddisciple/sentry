import {useState} from 'react';

import {Container, Flex, Stack} from '@sentry/scraps/layout';
import {Text} from '@sentry/scraps/text';

import * as Storybook from 'sentry/stories';
import {OrganizationContext} from 'sentry/utils/organizationContext';
import {useOrganization} from 'sentry/utils/useOrganization';
import {VisualizeTransformations} from './visualizeTransformations';

The span and log toolbars store each series' ordered transformations in URL query parameters.
The editor appears when the organization has the interpolation-and-smoothing feature.
Each series allows one Fill and one Smooth transformation, with two rows maximum.
When both functions are present, selecting the other type swaps their order and preserves their settings.

export function TransformationExample({layout = 'stacked'}) {
  const organization = useOrganization();
  const [transformations, setTransformations] = useState([
    'fill(zero)',
    'smooth(sma,10)',
  ]);
  return (
    <OrganizationContext
      value={{
        ...organization,
        features: [...organization.features, 'explore-interpolation-and-smoothing'],
      }}
    >
      <Container width={layout === 'inline' ? '100%' : '280px'}>
        <VisualizeTransformations
          transformations={transformations}
          onChange={values => setTransformations([...values])}
          layout={layout}
        >
          {layout === 'inline'
            ? ({addTrigger, rows}) => (
                <Stack gap="sm">
                  <Flex align="center" justify="between" gap="sm" wrap="wrap">
                    <Text size="sm" variant="muted">
                      900 data points
                    </Text>
                    {addTrigger}
                  </Flex>
                  {rows}
                </Stack>
              )
            : undefined}
        </VisualizeTransformations>
      </Container>
    </OrganizationContext>
  );
}

## Metrics layout

Metrics uses inline fields and places the Add Transformation trigger in the chart footer.
The render callback places the trigger and rows while the editor retains all transformation rules.

<Storybook.Demo>
  <TransformationExample layout="inline" />
</Storybook.Demo>

```tsx
<VisualizeTransformations
  transformations={transformations}
  onChange={setTransformations}
  layout="inline"
>
  {({addTrigger, rows}) => (
    <Stack gap="sm">
      <Flex justify="end">{addTrigger}</Flex>
      {rows}
    </Stack>
  )}
</VisualizeTransformations>
```

## Props

| Prop              | Type                                  | Description                                              |
| ----------------- | ------------------------------------- | -------------------------------------------------------- |
| `transformations` | `readonly string[]`                   | The series' ordered transformation expressions.          |
| `onChange`        | `(values: readonly string[]) => void` | Replaces the list through the parent query-state setter. |
| `layout`          | `'stacked' \| 'inline'`               | Arranges the fields. Defaults to `stacked`.              |
| `children`        | `({addTrigger, rows}) => ReactNode`   | Places the editor's trigger and rows.                    |

## Accessibility

Each transformation has a numbered group and labeled type, method, settings, and remove controls.
The selects use CompactSelect's keyboard navigation.
